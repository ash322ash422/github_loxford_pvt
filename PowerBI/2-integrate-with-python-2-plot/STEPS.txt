Go to the "Modeling" tab and create new measures:

Name
pl_count    = COUNTROWS(iris)
pl_mean = AVERAGE(iris[PetalLengthCm])
pl_minimum = MIN(iris[PetalLengthCm])
pl_maximum =  MAX(iris[PetalLengthCm])
pl_std_dev = STDEV.S(iris[PetalLengthCm])
pl_median = MEDIAN(iris[PetalLengthCm])
pl_q1 = PERCENTILE.INC(iris[PetalLengthCm],0.25)


pw_mean = AVERAGE(iris[PetalWidthCm])
